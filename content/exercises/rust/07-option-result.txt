fn parse_and_double(s: &str) -> Result<i32, String> {
    s.trim()
        .parse::<i32>()
        .map_err(|_| format!("not a number: {}", s))
        .map(|n| n * 2)
}

fn first_even(v: &[i32]) -> Option<i32> {
    v.iter().cloned().find(|n| n % 2 == 0)
}

fn main() {
    println!("{:?}", parse_and_double("21"));          // Ok(42)
    println!("{:?}", parse_and_double(" 8 "));          // Ok(16)
    println!("{:?}", parse_and_double("abc"));          // Err("not a number: abc")
    println!("{:?}", first_even(&[1, 3, 6, 8]));        // Some(6)
    println!("{:?}", first_even(&[1, 3]));              // None
}
