@@ solution
use std::fmt;
use std::num::ParseIntError;

#[derive(Debug)]
enum AgeError {
    NotANumber,
    OutOfRange(i32),
}

impl fmt::Display for AgeError {
    fn fmt(&self, f: &mut fmt::Formatter) -> fmt::Result {
        match self {
            AgeError::NotANumber => write!(f, "age must be a number"),
            AgeError::OutOfRange(n) => write!(f, "age out of range: {}", n),
        }
    }
}

// This is what lets `?` turn a ParseIntError into our AgeError.
impl From<ParseIntError> for AgeError {
    fn from(_: ParseIntError) -> AgeError {
        AgeError::NotANumber
    }
}

fn parse_age(text: &str) -> Result<i32, AgeError> {
    let n: i32 = text.trim().parse()?;
    if n < 0 || n > 150 {
        return Err(AgeError::OutOfRange(n));
    }
    Ok(n)
}

fn main() {
    for text in &["42", "abc", "-5", "200"] {
        match parse_age(text) {
            Ok(age) => println!("ok {}", age),
            Err(e) => println!("error: {}", e),
        }
    }
}
