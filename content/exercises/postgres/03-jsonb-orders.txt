@@ task
Each order is stored as one `jsonb` document. Write two queries:

1. The **customers whose total is above 200**, in alphabetical order. `doc->>'customer'` gives a field as text; cast the total with `(doc->>'total')::int` before comparing.
2. The **ids of the orders whose items include `book`**, in id order. `doc->'items' ? 'book'` tests whether a JSON array (or object) contains that string.
@@ hint
`SELECT doc->>'customer' FROM orders WHERE (doc->>'total')::int > 200 ORDER BY 1;` and `SELECT id FROM orders WHERE doc->'items' ? 'book' ORDER BY id;`
@@ expect
asha
meena
1
3
@@ starter
CREATE TABLE orders (id int PRIMARY KEY, doc jsonb NOT NULL);
INSERT INTO orders VALUES
  (1, '{"customer": "asha",  "total": 250, "items": ["pen", "book"]}'),
  (2, '{"customer": "ravi",  "total": 120, "items": ["ink"]}'),
  (3, '{"customer": "meena", "total": 300, "items": ["book", "bag"]}');

-- TODO 1: customers with a total above 200, alphabetical      -> asha, meena
-- TODO 2: ids of the orders whose items include "book"        -> 1, 3
@@ solution
CREATE TABLE orders (id int PRIMARY KEY, doc jsonb NOT NULL);
INSERT INTO orders VALUES
  (1, '{"customer": "asha",  "total": 250, "items": ["pen", "book"]}'),
  (2, '{"customer": "ravi",  "total": 120, "items": ["ink"]}'),
  (3, '{"customer": "meena", "total": 300, "items": ["book", "bag"]}');

SELECT doc->>'customer' AS customer FROM orders WHERE (doc->>'total')::int > 200 ORDER BY 1;
SELECT id FROM orders WHERE doc->'items' ? 'book' ORDER BY id;
