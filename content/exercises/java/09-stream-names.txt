@@ solution
import java.util.*;
import java.util.stream.*;

public class Main {
    static String shout(List<String> names) {
        return names.stream()
                .filter(n -> n.length() >= 4)
                .map(String::toUpperCase)
                .sorted()
                .collect(Collectors.joining(", "));
    }

    public static void main(String[] args) {
        System.out.println(shout(List.of("ravi", "li", "anita", "om", "zoya", "kabir")));   // ANITA, KABIR, RAVI, ZOYA
        System.out.println(shout(List.of("a", "bb", "cccc")));                              // CCCC
    }
}
