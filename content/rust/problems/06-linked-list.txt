struct Solution;

impl Solution {
    pub fn reorder_list(head: &mut Option<Box<ListNode>>) {
        let mut len = 0;
        let mut c = head.as_ref();
        while let Some(n) = c {
            len += 1;
            c = n.next.as_ref();
        }
        if len < 3 {
            return;
        }

        // 1. split: the first half keeps (len + 1) / 2 nodes
        let keep = (len + 1) / 2;
        let mut cur = head.as_mut().unwrap();
        for _ in 1..keep {
            cur = cur.next.as_mut().unwrap();
        }
        let second = cur.next.take();

        // 2. reverse the second half
        let mut prev = None;
        let mut c = second;
        while let Some(mut n) = c {
            c = n.next.take();
            n.next = prev;
            prev = Some(n);
        }
        let mut second = prev;

        // 3. interleave
        let mut first = head.take();
        let mut dummy = Box::new(ListNode { val: 0, next: None });
        let mut tail = &mut dummy;
        while first.is_some() || second.is_some() {
            if let Some(mut f) = first.take() {
                first = f.next.take();
                tail.next = Some(f);
                tail = tail.next.as_mut().unwrap();
            }
            if let Some(mut s) = second.take() {
                second = s.next.take();
                tail.next = Some(s);
                tail = tail.next.as_mut().unwrap();
            }
        }
        *head = dummy.next;
    }
}

struct Solution;

impl Solution {
    pub fn copy_random_list(nodes: &Vec<Node>, head: Option<usize>) -> (Vec<Node>, Option<usize>) {
        // pass 1: walk the list and give every node its new index
        let mut new_index: HashMap<usize, usize> = HashMap::new();
        let mut cur = head;
        while let Some(i) = cur {
            let id = new_index.len();
            new_index.insert(i, id);
            cur = nodes[i].next;
        }
        // pass 2: copy node by node, translating next / random through the map
        let mut copy = Vec::with_capacity(new_index.len());
        let mut cur = head;
        while let Some(i) = cur {
            let n = &nodes[i];
            copy.push(Node { val: n.val, next: n.next.map(|x| new_index[&x]), random: n.random.map(|x| new_index[&x]) });
            cur = n.next;
        }
        (copy, head.map(|h| new_index[&h]))
    }

S: A singly linked list is stored as a vector: `next[i]` is the index of the node after node `i`, or `-1` if node `i` is the tail. The head is node 0 (an empty vector is an empty list). Return whether the list contains a cycle.

A: Floyd's tortoise and hare: move one index by 1 and another by 2. If there is a cycle the fast index eventually laps the slow one; otherwise it reaches the `-1` tail. (A cycle cannot be built from owning `Box` nodes, which is why the list is index-based here.)

impl Solution {
    pub fn has_cycle(next: Vec<i32>) -> bool {
        if next.is_empty() {
            return false;
        }
        let (mut slow, mut fast) = (0i32, 0i32);
        loop {
            for _ in 0..2 {
                if fast < 0 {
                    return false;
                }
                fast = next[fast as usize];
            }
            if fast < 0 {
                return false;
            }
            slow = next[slow as usize];
            if slow == fast {
                return true;
            }
        }

S: A singly linked list is stored as a vector: `next[i]` is the index of the node after node `i`, or `-1` for the tail; the head is node 0. If the list has a cycle, return the index of the node where the cycle begins, otherwise `-1`.

A: Use Floyd's algorithm to detect a meeting point. Then restart one index from the head and move both one step at a time; they meet exactly at the cycle entrance, because the distance from the head to the entrance equals the distance from the meeting point to it around the loop.
C: Solution::detect_cycle(vec![1,2,3,1]) == 1 && Solution::detect_cycle(vec![1,-1]) == -1 && Solution::detect_cycle(vec![]) == -1 && Solution::detect_cycle(vec![0]) == 0

struct Solution;

impl Solution {
    pub fn detect_cycle(next: Vec<i32>) -> i32 {
        if next.is_empty() {
            return -1;
        }
        let (mut slow, mut fast) = (0i32, 0i32);
        loop {
            for _ in 0..2 {
                if fast < 0 {
                    return -1;
                }
                fast = next[fast as usize];
            }
            if fast < 0 {
                return -1;
            }
            slow = next[slow as usize];
            if slow == fast {
                let mut entry = 0i32; // met inside the cycle
                while entry != slow {
                    entry = next[entry as usize];
                    slow = next[slow as usize];
                }
                return entry;
            }
        }
    }

A: A `HashMap` gives O(1) lookup of nodes; a doubly linked list keeps usage order (most recent at the front). Safe Rust cannot hold two owning pointers per node, so the list lives in a `Vec` arena and nodes link by index; evicted slots are recycled through a free list. Every access moves the node to the front; eviction removes the node before the tail sentinel.
C: { let mut c = LRUCache::new(2); c.put(1, 1); c.put(2, 2); let a = c.get(1) == 1; c.put(3, 3); let b = c.get(2) == -1; c.put(4, 4); a && b && c.get(1) == -1 && c.get(3) == 3 && c.get(4) == 4 }
```rust
use std::collections::HashMap;

struct LRUCache {
    cap: usize,
    map: HashMap<i32, usize>, // key -> index in `nodes`
    nodes: Vec<Node>,         // nodes[0] = head sentinel, nodes[1] = tail sentinel
    free: Vec<usize>,         // recycled slots
}

struct Node {
    key: i32,
    val: i32,
    prev: usize,
    next: usize,
}

impl LRUCache {
    pub fn new(capacity: i32) -> Self {
        let head = Node { key: 0, val: 0, prev: 0, next: 1 };
        let tail = Node { key: 0, val: 0, prev: 0, next: 1 };
        LRUCache { cap: capacity as usize, map: HashMap::new(), nodes: vec![head, tail], free: Vec::new() }
    }

    fn unlink(&mut self, i: usize) {
        let (p, n) = (self.nodes[i].prev, self.nodes[i].next);
        self.nodes[p].next = n;
        self.nodes[n].prev = p;
    }

    fn push_front(&mut self, i: usize) {
        let first = self.nodes[0].next;
        self.nodes[i].prev = 0;
        self.nodes[i].next = first;
        self.nodes[first].prev = i;
        self.nodes[0].next = i;
    }

    pub fn get(&mut self, key: i32) -> i32 {
        match self.map.get(&key).copied() {
            Some(i) => {
                self.unlink(i);
                self.push_front(i); // now the most recently used
                self.nodes[i].val
            }
            None => -1,
        }
    }

    pub fn put(&mut self, key: i32, value: i32) {
        if let Some(&i) = self.map.get(&key) {
            self.nodes[i].val = value;
            self.unlink(i);
            self.push_front(i);
            return;
        }
        if self.map.len() == self.cap {
            let lru = self.nodes[1].prev; // least recently used
            self.unlink(lru);
            let k = self.nodes[lru].key;
            self.map.remove(&k);
            self.free.push(lru);
        }
        let node = Node { key, val: value, prev: 0, next: 0 };
        let i = match self.free.pop() {
            Some(i) => {
                self.nodes[i] = node;
                i
            }
            None => {
                self.nodes.push(node);
                self.nodes.len() - 1
            }
        };
        self.map.insert(key, i);
        self.push_front(i);
    }
}
```

use std::cmp::Reverse;
use std::collections::BinaryHeap;

struct Solution;

impl Solution {
    pub fn merge_k_lists(lists: Vec<Option<Box<ListNode>>>) -> Option<Box<ListNode>> {
        let mut lists = lists;
        let mut heap = BinaryHeap::new(); // min-heap of (value, list index)
        for (i, l) in lists.iter().enumerate() {
            if let Some(n) = l {
                heap.push(Reverse((n.val, i)));
            }
        }
        let mut dummy = Box::new(ListNode { val: 0, next: None });
        let mut tail = &mut dummy;
        while let Some(Reverse((_, i))) = heap.pop() {
            let mut node = lists[i].take().unwrap();
            lists[i] = node.next.take();
            if let Some(n) = &lists[i] {
                heap.push(Reverse((n.val, i)));
            }
            tail.next = Some(node);
            tail = tail.next.as_mut().unwrap();
        }
        dummy.next
    }
}
