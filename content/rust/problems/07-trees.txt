```rust
    fn height(node: &Option<Box<TreeNode>>) -> i32 {
```

```rust
impl Solution {
    pub fn lowest_common_ancestor(root: &Option<Box<TreeNode>>, p: i32, q: i32) -> i32 {
        let mut node = root.as_ref();
        while let Some(n) = node {
            if p < n.val && q < n.val {
                node = n.left.as_ref();
            } else if p > n.val && q > n.val {
                node = n.right.as_ref();
            } else {
                return n.val; // split point (or one of them)
            }

    }
```

A: Breadth-first search with a `VecDeque`. Process exactly `queue.len()` nodes per round so each round is one level, pushing children for the next round.
C: Solution::level_order(&tree("[3,9,20,null,null,15,7]")) == vec![vec![3], vec![9,20], vec![15,7]] && Solution::level_order(&tree("[1]")) == vec![vec![1]] && Solution::level_order(&None).is_empty()

```rust
use std::collections::VecDeque;

struct Solution;

impl Solution {
    pub fn level_order(root: &Option<Box<TreeNode>>) -> Vec<Vec<i32>> {
        let mut result = Vec::new();
        let mut queue: VecDeque<&TreeNode> = VecDeque::new();
        if let Some(r) = root {
            queue.push_back(r);
        }
        while !queue.is_empty() {
            let mut level = Vec::new();
            for _ in 0..queue.len() {
                // nodes on this level
                let node = queue.pop_front().unwrap();
                level.push(node.val);
                if let Some(l) = &node.left {
                    queue.push_back(l);
                }
                if let Some(r) = &node.right {
                    queue.push_back(r);
                }
            }
            result.push(level);
        }
        result
    }
}
```

```rust
struct Solution;

impl Solution {
    pub fn kth_smallest(root: &Option<Box<TreeNode>>, k: i32) -> i32 {
        let mut stack: Vec<&TreeNode> = Vec::new();
        let mut node = root.as_deref();
        let mut k = k;
        while node.is_some() || !stack.is_empty() {
            while let Some(n) = node {
                // go as far left as possible
                stack.push(n);
                node = n.left.as_deref();
            }
            let n = stack.pop().unwrap();
            k -= 1;
            if k == 0 {
                return n.val; // the k-th in-order node
            }
            node = n.right.as_deref();
        }
        -1
    }
}
```

```rust
struct Solution;

impl Solution {
    pub fn build_tree(preorder: Vec<i32>, inorder: Vec<i32>) -> Option<Box<TreeNode>> {
        let index: HashMap<i32, usize> = inorder.iter().enumerate().map(|(i, &v)| (v, i)).collect();
        let mut pre = 0;
        Self::build(&preorder, &index, &mut pre, 0, inorder.len() as isize - 1)
    }

    fn build(preorder: &[i32], index: &HashMap<i32, usize>, pre: &mut usize, lo: isize, hi: isize) -> Option<Box<TreeNode>> {
        if lo > hi {
            return None;
        }
        let val = preorder[*pre];
        *pre += 1;
        let mid = index[&val] as isize; // split point in inorder
        let left = Self::build(preorder, index, pre, lo, mid - 1);
        let right = Self::build(preorder, index, pre, mid + 1, hi);
        Some(Box::new(TreeNode { val, left, right }))
    }
}
```

```rust
struct TreeCodec;

impl TreeCodec {
    pub fn new() -> Self {
        TreeCodec
    }

    pub fn serialize(&self, root: &Option<Box<TreeNode>>) -> String {
        let mut out: Vec<String> = Vec::new();
        Self::write(root, &mut out);
        out.join(",")
    }

    fn write(node: &Option<Box<TreeNode>>, out: &mut Vec<String>) {
        match node {
            None => out.push("#".to_string()),
            Some(n) => {
                out.push(n.val.to_string());
                Self::write(&n.left, out);
                Self::write(&n.right, out);
            }
        }
    }

    pub fn deserialize(&self, data: String) -> Option<Box<TreeNode>> {
        let mut tokens = data.split(',');
        Self::read(&mut tokens)
    }

    fn read(tokens: &mut std::str::Split<char>) -> Option<Box<TreeNode>> {
        match tokens.next() {
            None | Some("#") => None,
            Some(t) => {
                let val: i32 = t.parse().unwrap();
                let left = Self::read(tokens);
                let right = Self::read(tokens);
                Some(Box::new(TreeNode { val, left, right }))
            }
        }
    }
}
```
