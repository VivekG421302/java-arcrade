=== 21
A: Track the lowest price seen so far. For each day, the best profit selling today is `price - min_price`. Keep the maximum. One pass, O(1) space.
C: Solution::max_profit(vec![7,1,5,3,6,4]) == 5 && Solution::max_profit(vec![7,6,4,3,1]) == 0 && Solution::max_profit(vec![2,4,1]) == 2
```rust
struct Solution;

impl Solution {
    pub fn max_profit(prices: Vec<i32>) -> i32 {
        let mut min_price = std::i32::MAX;
        let mut best = 0;
        for p in prices {
            min_price = min_price.min(p);
            best = best.max(p - min_price);
        }
        best
    }
}
```
=== 22
A: Sliding window with a table of each byte's last seen index. When a character repeats inside the window, jump the window start just past its previous position. The window always holds unique characters.
C: Solution::length_of_longest_substring("abcabcbb".to_string()) == 3 && Solution::length_of_longest_substring("bbbbb".to_string()) == 1 && Solution::length_of_longest_substring("pwwkew".to_string()) == 3 && Solution::length_of_longest_substring("".to_string()) == 0
```rust
struct Solution;

impl Solution {
    pub fn length_of_longest_substring(s: String) -> i32 {
        let mut last = [-1i32; 128];
        let (mut start, mut best) = (0i32, 0i32);
        for (i, b) in s.bytes().enumerate() {
            let i = i as i32;
            if last[b as usize] >= start {
                start = last[b as usize] + 1; // repeat inside window: shrink
            }
            last[b as usize] = i;
            best = best.max(i - start + 1);
        }
        best
    }
}
```
=== 23
A: Keep a window and the count of its most frequent letter. The window is valid while `length - max_count <= k` (the rest can be replaced). Otherwise slide the left edge. The max count never needs to shrink, because only a larger count can improve the answer.
C: Solution::character_replacement("ABAB".to_string(), 2) == 4 && Solution::character_replacement("AABABBA".to_string(), 1) == 4 && Solution::character_replacement("AAAA".to_string(), 0) == 4
```rust
struct Solution;

impl Solution {
    pub fn character_replacement(s: String, k: i32) -> i32 {
        let b = s.as_bytes();
        let mut count = [0i32; 26];
        let (mut left, mut max_count, mut best) = (0usize, 0i32, 0i32);
        for right in 0..b.len() {
            count[(b[right] - b'A') as usize] += 1;
            max_count = max_count.max(count[(b[right] - b'A') as usize]);
            while (right - left + 1) as i32 - max_count > k {
                count[(b[left] - b'A') as usize] -= 1; // too many replacements needed
                left += 1;
            }
            best = best.max((right - left + 1) as i32);
        }
        best
    }
}
```
=== 24
A: Slide a fixed window of length `s1.len()` over `s2`, maintaining letter counts for the window and comparing with the counts of `s1`. Adding the new letter and removing the old one is O(1) per step.
C: Solution::check_inclusion("ab".to_string(), "eidbaooo".to_string()) && !Solution::check_inclusion("ab".to_string(), "eidboaoo".to_string()) && !Solution::check_inclusion("abc".to_string(), "ab".to_string())
```rust
struct Solution;

impl Solution {
    pub fn check_inclusion(s1: String, s2: String) -> bool {
        let (a, b) = (s1.as_bytes(), s2.as_bytes());
        if a.len() > b.len() {
            return false;
        }
        let mut need = [0i32; 26];
        let mut window = [0i32; 26];
        for i in 0..a.len() {
            need[(a[i] - b'a') as usize] += 1;
            window[(b[i] - b'a') as usize] += 1;
        }
        if need == window {
            return true;
        }
        for i in a.len()..b.len() {
            window[(b[i] - b'a') as usize] += 1; // enter
            window[(b[i - a.len()] - b'a') as usize] -= 1; // leave
            if need == window {
                return true;
            }
        }
        false
    }
}
```

=== 25
A: Expand the right edge until the window covers all of `t` (tracked with a `missing` counter and a need table). Then shrink from the left while it still covers everything, recording the best window each time.
C: Solution::min_window("ADOBECODEBANC".to_string(), "ABC".to_string()) == "BANC" && Solution::min_window("a".to_string(), "a".to_string()) == "a" && Solution::min_window("a".to_string(), "aa".to_string()) == ""
```rust
struct Solution;

impl Solution {
    pub fn min_window(s: String, t: String) -> String {
        let (sb, tb) = (s.as_bytes(), t.as_bytes());
        if tb.len() > sb.len() {
            return String::new();
        }
        let mut need = [0i32; 128];
        for &c in tb {
            need[c as usize] += 1;
        }
        let mut missing = tb.len();
        let (mut left, mut best_start, mut best_len) = (0usize, 0usize, std::usize::MAX);
        for right in 0..sb.len() {
            let c = sb[right] as usize;
            if need[c] > 0 {
                missing -= 1; // a needed char entered
            }
            need[c] -= 1;
            while missing == 0 {
                // window is valid: try to shrink
                if right - left + 1 < best_len {
                    best_len = right - left + 1;
                    best_start = left;
                }
                let lc = sb[left] as usize;
                need[lc] += 1;
                if need[lc] > 0 {
                    missing += 1; // a needed char left
                }
                left += 1;
            }
        }
        if best_len == std::usize::MAX {
            String::new()
        } else {
            s[best_start..best_start + best_len].to_string()
        }
    }
}
```
=== 26
A: Keep a `VecDeque` of indices whose values are in decreasing order. Before adding index `i`, drop smaller values from the back (they can never be a max again) and drop the front if it left the window. The front is always the current maximum.
C: Solution::max_sliding_window(vec![1,3,-1,-3,5,3,6,7], 3) == vec![3,3,5,5,6,7] && Solution::max_sliding_window(vec![1], 1) == vec![1] && Solution::max_sliding_window(vec![9,11], 2) == vec![11]
```rust
use std::collections::VecDeque;

struct Solution;

impl Solution {
    pub fn max_sliding_window(nums: Vec<i32>, k: i32) -> Vec<i32> {
        let k = k as usize;
        let mut result = Vec::new();
        let mut dq: VecDeque<usize> = VecDeque::new(); // indices, values decreasing
        for i in 0..nums.len() {
            if let Some(&front) = dq.front() {
                if front + k <= i {
                    dq.pop_front(); // out of window
                }
            }
            while let Some(&back) = dq.back() {
                if nums[back] < nums[i] {
                    dq.pop_back(); // useless smaller value
                } else {
                    break;
                }
            }
            dq.push_back(i);
            if i + 1 >= k {
                result.push(nums[*dq.front().unwrap()]);
            }
        }
        result
    }
}
```
=== 27
A: Compute the sum of the first `k` elements, then slide: add the new element, subtract the one leaving. Track the maximum sum and divide by `k` once at the end.
C: (Solution::find_max_average(vec![1,12,-5,-6,50,3], 4) - 12.75).abs() < 1e-5 && (Solution::find_max_average(vec![5], 1) - 5.0).abs() < 1e-5
```rust
struct Solution;

impl Solution {
    pub fn find_max_average(nums: Vec<i32>, k: i32) -> f64 {
        let k = k as usize;
        let mut sum: i64 = nums[..k].iter().map(|&x| x as i64).sum();
        let mut best = sum;
        for i in k..nums.len() {
            sum += (nums[i] - nums[i - k]) as i64; // slide the window by one
            best = best.max(sum);
        }
        best as f64 / k as f64
    }
}
```
