```rust
struct Solution;

impl Solution {
    pub fn combination_sum(candidates: Vec<i32>, target: i32) -> Vec<Vec<i32>> {
        let mut res = Vec::new();
        Self::dfs(&candidates, 0, target, &mut Vec::new(), &mut res);
        res
    }

    fn dfs(c: &[i32], start: usize, remain: i32, cur: &mut Vec<i32>, res: &mut Vec<Vec<i32>>) {
        if remain == 0 {
            res.push(cur.clone());
            return;
        }
        for i in start..c.len() {
            if c[i] > remain {
                continue;
            }
            cur.push(c[i]);
            Self::dfs(c, i, remain - c[i], cur, res); // i, not i+1: reuse allowed
            cur.pop();
        }
    }
```

```rust
            return;

                continue;
```

```rust
struct Solution;

impl Solution {
    pub fn exist(board: Vec<Vec<char>>, word: String) -> bool {
        let mut board = board;
        let w: Vec<char> = word.chars().collect();
        for r in 0..board.len() as i32 {
            for c in 0..board[0].len() as i32 {
                if Self::dfs(&mut board, &w, 0, r, c) {
                    return true;
                }
            }
        }
        false
    }

    fn dfs(b: &mut Vec<Vec<char>>, w: &[char], k: usize, r: i32, c: i32) -> bool {
        if k == w.len() {
            return true;
        }
        if r < 0 || c < 0 || r as usize >= b.len() || c as usize >= b[0].len() || b[r as usize][c as usize] != w[k] {
            return false;
        }
        let saved = b[r as usize][c as usize];
        b[r as usize][c as usize] = '#'; // mark as used
        let found = Self::dfs(b, w, k + 1, r + 1, c)
            || Self::dfs(b, w, k + 1, r - 1, c)
            || Self::dfs(b, w, k + 1, r, c + 1)
            || Self::dfs(b, w, k + 1, r, c - 1);
        b[r as usize][c as usize] = saved; // backtrack
        found
    }
}
```

```rust
struct Solution;

impl Solution {
    pub fn partition(s: String) -> Vec<Vec<String>> {
        let mut res = Vec::new();
        Self::dfs(s.as_bytes(), 0, &mut Vec::new(), &mut res);
        res
    }

    fn dfs(s: &[u8], start: usize, cur: &mut Vec<String>, res: &mut Vec<Vec<String>>) {
        if start == s.len() {
            res.push(cur.clone());
            return;
        }
        for end in start..s.len() {
            if Self::is_palindrome(&s[start..=end]) {
                cur.push(String::from_utf8(s[start..=end].to_vec()).unwrap());
                Self::dfs(s, end + 1, cur, res);
                cur.pop();
            }
        }
    }

    fn is_palindrome(s: &[u8]) -> bool {
        s.iter().eq(s.iter().rev())
    }
}
```

A: Place one queen per row, trying each column. Track used columns and both diagonals in boolean vectors (`row - col` and `row + col` identify diagonals) so each check is O(1). Backtrack when no column is free.
C: Solution::solve_n_queens(4) == vec![vec![".Q..", "...Q", "Q...", "..Q."], vec!["..Q.", "Q...", "...Q", ".Q.."]] && Solution::solve_n_queens(1) == vec![vec!["Q"]] && Solution::solve_n_queens(6).len() == 4 && Solution::solve_n_queens(8).len() == 92
```rust
struct Solution;

impl Solution {
    pub fn solve_n_queens(n: i32) -> Vec<Vec<String>> {
        let n = n as usize;
        let mut res = Vec::new();
        let mut cols = vec![false; n];
        let mut diag = vec![false; 2 * n];
        let mut anti = vec![false; 2 * n];
        let mut queen_col = vec![0usize; n];
        Self::place(0, n, &mut queen_col, &mut cols, &mut diag, &mut anti, &mut res);
        res
    }

    fn place(row: usize, n: usize, queen_col: &mut Vec<usize>, cols: &mut Vec<bool>, diag: &mut Vec<bool>, anti: &mut Vec<bool>, res: &mut Vec<Vec<String>>) {
        if row == n {
            res.push(Self::render(queen_col, n));
            return;
        }
        for c in 0..n {
            let (d, a) = (row + n - c, row + c); // diagonal ids
            if cols[c] || diag[d] || anti[a] {
                continue;
            }
            cols[c] = true;
            diag[d] = true;
            anti[a] = true;
            queen_col[row] = c;
            Self::place(row + 1, n, queen_col, cols, diag, anti, res);
            cols[c] = false; // backtrack
            diag[d] = false;
            anti[a] = false;
        }
    }

    fn render(queen_col: &[usize], n: usize) -> Vec<String> {
        (0..n)
            .map(|r| {
                let mut line = vec!['.'; n];
                line[queen_col[r]] = 'Q';
                line.into_iter().collect()
            })
            .collect()
    }
}
```
